{{- /* social-meta.html Replaces Hugo's `_internal/opengraph.html` + `_internal/twitter_cards.html`, which never emitted an og:image for this site: they look for `.Params.images`, a page-bundle resource whose name contains "feature" or "cover", or `site.Params.images` — and none of those exist here. Posts reference their hero from the markdown body, so it has to be recovered from the page source. Image resolution, first match wins: 1. `images` in front matter (explicit override) 2. the post's hero image, read from the markdown source and rendered to a purpose-built 1200x630 JPEG 3. site-wide fallback: the generated `og_image` card Deliberately JPEG, not WebP. The image render hook emits WebP for the page itself, but LinkedIn in particular is unreliable at unfurling WebP cards, so the social rendition is a separate, conservatively-encoded derivative. The hero path is taken from `.RawContent` rather than the rendered `.Content` precisely so it resolves to the original asset and not the WebP variant. */ -}} {{- $imgAbs := "" -}} {{- $ogW := 1200 -}} {{- $ogH := 630 -}} {{- /* 1. explicit override */ -}} {{- with .Params.images -}} {{- $imgAbs = (index . 0) | absURL -}} {{- end -}} {{- /* 2. first markdown image in the page source, filled to card dimensions */ -}} {{- if and (not $imgAbs) .IsPage -}} {{- $path := "" -}} {{- with findRE `!\[[^\]]*\]\([^)\s]+` .RawContent 1 -}} {{- $path = replaceRE `^!\[[^\]]*\]\(` "" (index . 0) -}} {{- end -}} {{- if and $path (not (strings.HasPrefix $path "http")) -}} {{- with resources.Get (strings.TrimPrefix "/" $path) -}} {{- $card := .Fill (printf "%dx%d jpg q85 Center" $ogW $ogH) -}} {{- $imgAbs = $card.Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 3. sitewide default card */ -}} {{- if not $imgAbs -}} {{- with site.Params.og_image -}}{{- $imgAbs = . | absURL -}}{{- end -}} {{- end -}} {{- $desc := "" -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- if .IsPage -}} {{- $desc = .Summary | plainify | truncate 200 -}} {{- else -}} {{- $desc = site.Params.description -}} {{- end -}} {{- end -}} {{- with $imgAbs }} {{- /* Every card is generated at exactly these dimensions, whether it is a post hero filled to size or the sitewide default, so the declared width/height is always accurate. */ -}} {{- end }} {{- if .IsPage }} {{- if not .Date.IsZero }} {{- end }} {{- if not .Lastmod.IsZero }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }} {{- /* `summary` renders as a small square thumbnail; `summary_large_image` is what makes a shared link read as a card. */ -}} {{- with $imgAbs }} {{- end }} {{- with site.Params.twitter_handle }} {{- end }}
