{{- /* Link render hook. 21 of 120 notes end with their source as a bare URL on its own line. Goldmark autolinks those, so they were clickable — but they rendered at full length, wrapping across three lines mid-card and mid-paragraph. When the link text is identical to its destination (i.e. it was autolinked rather than written as [text](url)), show the domain instead. The full URL stays in href and in the title attribute, so nothing is hidden. Hand-written links are passed through untouched. */ -}} {{- $dest := .Destination -}} {{- $text := .Text -}} {{- $isBare := and (eq (trim $text " ") (trim $dest " ")) (strings.HasPrefix $dest "http") -}} {{- $external := strings.HasPrefix $dest "http" -}} {{- if $isBare -}} {{- $domain := replaceRE `^https?://(www\.)?([^/]+).*$` "$2" $dest -}} {{ $domain }} ↗ {{- else if $external -}} {{ $text | safeHTML }} {{- else -}} {{ $text | safeHTML }} {{- end -}}
